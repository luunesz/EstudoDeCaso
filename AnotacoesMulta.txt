CREATE TABLE IF NOT EXISTS Aluno (

    placa CHAR(7) PRIMARY KEY,

    fk_Viagem_id CHAR(10),

CREATE TABLE IF NOT EXISTS inscrito (
    fk_Aluno_cpf CHAR(11),

CREATE TABLE IF NOT EXISTS Responsavel_Aluno (
    fk_Responsavel_cpf CHAR(11),

-- Função para contar presenças do aluno de acordo com a série
CREATE OR REPLACE FUNCTION contar_presencas(
    p_cpf CHAR(11),
    p_serie VARCHAR
)
RETURNS INTEGER AS $$
DECLARE
    v_total INTEGER;
BEGIN
    SELECT COUNT(*)
    INTO v_total
    FROM presenca p
    JOIN aluno a ON a.cpf = p.fk_aluno_cpf
    WHERE a.cpf = p_cpf
      AND a.serie = p_serie;

    RETURN v_total;
END;
$$ LANGUAGE plpgsql;


-- Procedure para registrar multa e atualizar status do motorista
CREATE OR REPLACE PROCEDURE registrar_multa(
    p_id CHAR(10), p_viagem_id CHAR(10),
    p_tipo VARCHAR(50), p_valor FLOAT
)
LANGUAGE plpgsql AS $$
DECLARE
    v_cnh CHAR(11); 
    v_total INTEGER;
BEGIN
    SELECT fk_Motorista_cnh INTO v_cnh
    FROM Viagem WHERE id = p_viagem_id;
    
    IF v_cnh IS NULL THEN
        RAISE EXCEPTION 'Viagem % não encontrada ou sem motorista.', p_viagem_id;
    END IF;

    INSERT INTO Multa (id, data_emissao, valor_multa, status_multa, fk_Viagem_id, tipo)
    VALUES (p_id, CURRENT_DATE, p_valor, 'pendente', p_viagem_id, p_tipo);

    UPDATE Motorista SET num_multa = num_multa + 1
    WHERE cnh = v_cnh
    RETURNING num_multa INTO v_total;

    IF v_total >= 3 THEN
        UPDATE Viagem SET status_viagem = 'suspensa'
        WHERE fk_Motorista_cnh = v_cnh
          AND status_viagem = 'agendada';
    END IF;
END;
$$;

CREATE OR REPLACE VIEW vw_dash_multa AS
SELECT 
    tipo,
    status_multa,
    valor_multa,
    data_emissao,
    TO_CHAR(data_emissao, 'YYYY-MM') AS mes_emissao
FROM Multa;

      ('R001', 'Escolar', 1, 2, 'Ativa')

      ('12345678901', 'Carlos Silva', 40, 'Masculino', 0)

      ('ABC1234', 'Amarelo', 12.5, 45, 'Em Operacao')
    ON CONFLICT (placa) DO NOTHING;

    -- 2. CRIAR 10 VIAGENS DE EXEMPLO
    FOR i IN 1..10 LOOP
        INSERT INTO Viagem (id, local_saida, local_chegada, fk_Rota_id, fk_Motorista_cnh, fk_Onibus_placa, status_viagem)
        VALUES (
            'V' || lpad(i::text, 4, '0'),
            'Origem ' || i,
            'Destino ' || i,
            'R001',
            '12345678901',
            'ABC1234',
            'concluida'
        )
        ON CONFLICT (id) DO NOTHING;
    END LOOP;

    -- 3. COLETAR AS VIAGENS PARA LIGAR ÀS MULTAS
    SELECT ARRAY_AGG(id) INTO v_viagens FROM Viagem;

    -- 4. GERAR AS 1.000 MULTAS
    FOR i IN 1..1000 LOOP
        v_viagem_id := v_viagens[1 + floor(random() * array_length(v_viagens, 1))];

        VALUES (
            'M' || lpad(i::text, 5, '0'), 
            v_data, 
            v_valor, 
            v_status_multa, 
            v_viagem_id, 
            v_tipo
        )
        ON CONFLICT (id) DO NOTHING;
    END LOOP;
    
    RAISE NOTICE 'Dados base e 1.000 multas geradas com sucesso!';
END;
$$;
