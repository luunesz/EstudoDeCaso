CREATE VIEW vw_dash_presenca AS
SELECT p.fk_viagem_id AS viagem, p.data_presenca, a.serie
FROM presenca p
JOIN aluno a ON a.cpf = p.fk_aluno_cpf;

GRANT SELECT ON vw_dash_presenca TO anon;

CREATE VIEW vw_dash_aluno AS
SELECT a.serie, COUNT(p.fk_viagem_id) AS total_presencas
FROM aluno a
LEFT JOIN presenca p ON p.fk_aluno_cpf = a.cpf
WHERE a.cpf IN (SELECT fk_aluno_cpf FROM inscrito)
GROUP BY a.cpf, a.serie;

GRANT SELECT ON vw_dash_aluno TO anon;
